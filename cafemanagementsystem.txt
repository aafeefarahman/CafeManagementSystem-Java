import javax.swing.*;
import javax.swing.border.TitledBorder;
import java.awt.*;

import java.time.LocalDateTime;

public class CafeManagementSystem extends JFrame {
    // 🔴 ABSOLUTE BASE PATH (FIXED)
    static final String BASE =
            "C:\\Users\\tulas\\OneDrive\\Desktop\\CafeManagmentSystem\\";
    static final String BG_IMAGE = BASE + "background.png";
    // ===== IMAGE LOADER =====
    private ImageIcon loadImage(String path, int w, int h) {
        File f = new File(path);
        if (!f.exists()) {
            System.out.println("❌ Image NOT found: " + path);
            return null;
        }
        Image img = new ImageIcon(path).getImage()
                .getScaledInstance(w, h, Image.SCALE_SMOOTH);
        return new ImageIcon(img);
    }
    // ===== FIND UPI QR IMAGE (png / jpg / jpeg) =====
    private String getUPIQRPath() {
        String[] possible = {
                BASE + "upi_qr.png",
                BASE + "upi_qr.jpg",
                BASE + "upi_qr.jpg"
        };
        for (String p : possible) {
            if (new File(p).exists()) {
                System.out.println("✅ QR FOUND: " + p);
                return p;
            }
        }
        System.out.println("❌ No QR image found in folder");
        return null;
    }
    // ===== BACKGROUND PANEL =====
    class BackgroundPanel extends JPanel {
        Image bg;
        BackgroundPanel(String path) {
            if (new File(path).exists())
                bg = new ImageIcon(path).getImage();
            setLayout(null);
        }
        protected void paintComponent(Graphics g) {
            super.paintComponent(g);
            if (bg != null)
                g.drawImage(bg, 0, 0, getWidth(), getHeight(), this);
        }
    }
    // ===== MENU ITEM =====
    static class MenuItem {
        String name, imageFile;
        int price, qty = 0;
        JLabel qtyLabel;
        MenuItem(String n, String img, int p) {
            name = n;
            imageFile = img;
            price = p;
        }
    }
    private final List<MenuItem> items = new ArrayList<>();
    private JTextArea receipt;
    private JRadioButton cash, upi, card;

        setContentPane(new BackgroundPanel(BG_IMAGE));

        add(title);
        // ===== MENU ITEMS =====
        addItem("Tea", "tea.png", 30);

        addItem("Juice", "juice.png", 60);
        addItem("Pizza", "pizza.png", 150);
        addItem("Burger", "burger.png", 120);
        addItem("Sandwich", "sandwich.png", 80);
        addItem("Pasta", "pasta.png", 100);
        addItem("Shakes", "shakes.png", 90);
        addItem("Donut", "donut.png", 40);

        addItem("Ice Cream", "icecream.png", 60);

        // ===== MENU PANEL =====

        menuTitle.setFont(new Font("Arial", Font.BOLD, 24));
        menuTitle.setOpaque(true);
        menuTitle.setBackground(Color.BLACK);
        menuTitle.setForeground(Color.WHITE);
        menuPanel.add(menuTitle, BorderLayout.NORTH);
        JPanel grid = new JPanel(new GridLayout(0, 3, 20, 20));

    private void addItem(String name, String img, int price) {
        items.add(new MenuItem(name, BASE + img, price));
    }
    private JPanel createItemCard(MenuItem item) {
        JPanel card = new JPanel(new BorderLayout());
        card.setPreferredSize(new Dimension(230, 260));
        card.setBackground(Color.WHITE);
        card.setBorder(BorderFactory.createLineBorder(Color.GRAY));
        JLabel imgLabel = new JLabel("", SwingConstants.CENTER);
        ImageIcon icon = loadImage(item.imageFile, 160, 120);
        if (icon != null) imgLabel.setIcon(icon);
        JLabel name = new JLabel(item.name + "  ₹" + item.price, SwingConstants.CENTER);
        name.setFont(new Font("Arial", Font.BOLD, 14));
        JButton plus = new JButton("+");
        JButton minus = new JButton("-");
        item.qtyLabel = new JLabel("0", SwingConstants.CENTER);
        plus.addActionListener(e -> {
            item.qty++;
            item.qtyLabel.setText(String.valueOf(item.qty));
        });
        minus.addActionListener(e -> {
            if (item.qty > 0) {
                item.qty--;
                item.qtyLabel.setText(String.valueOf(item.qty));
            }
        });
        JPanel bottom = new JPanel();
        bottom.add(minus);
        bottom.add(item.qtyLabel);
        bottom.add(plus);
        card.add(imgLabel, BorderLayout.NORTH);
        card.add(name, BorderLayout.CENTER);
        card.add(bottom, BorderLayout.SOUTH);
        return card;
    }
    private void generateBill() {
        int total = 0;
        StringBuilder sb = new StringBuilder();
        sb.append("===== Cafe Receipt =====\n");
        sb.append(LocalDateTime.now()
                .format(DateTimeFormatter.ofPattern("dd-MM-yyyy HH:mm:ss")))
          .append("\n-------------------------\n");
        for (MenuItem i : items) {
            if (i.qty > 0) {
                int p = i.qty * i.price;
                sb.append(i.name).append(" x").append(i.qty)
                  .append(" = ₹").append(p).append("\n");
                total += p;
            }
        }
        sb.append("-------------------------\n");
        sb.append("Total: ₹").append(total).append("\n");
        sb.append("Payment: ")
          .append(upi.isSelected() ? "UPI" : card.isSelected() ? "Card" : "Cash")
          .append("\n\nThank You! Visit Again ☕");
        receipt.setText(sb.toString());
        if (upi.isSelected()) showUPIQR();
    }
    private void showUPIQR() {
        String qrPath = getUPIQRPath();
        JDialog d = new JDialog(this, "Scan to Pay (UPI)", true);
        d.setSize(420, 480);
        d.setLocationRelativeTo(this);
        JLabel qr = new JLabel("", SwingConstants.CENTER);
        if (qrPath != null) {
            qr.setIcon(loadImage(qrPath, 300, 300));
        } else {
            qr.setText("QR IMAGE NOT FOUND");
            qr.setFont(new Font("Arial", Font.BOLD, 16));
        }
        d.add(qr);
        d.setVisible(true);
    }
    private void resetAll() {
        for (MenuItem i : items) {
            i.qty = 0;
            i.qtyLabel.setText("0");
        }
        receipt.setText("");
        cash.setSelected(true);
    }
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> new CafeManagementSystem().setVisible(true));
}
}
